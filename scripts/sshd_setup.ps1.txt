$ErrorActionPreference = 'Stop'
Set-Service sshd -StartupType Automatic
Stop-Service sshd -Force -ErrorAction SilentlyContinue
(Get-Content C:\ProgramData\ssh\sshd_config) -replace '^#?\s*Port\s+.*', 'Port __PORT__' | Set-Content C:\ProgramData\ssh\sshd_config
(Get-Content C:\ProgramData\ssh\sshd_config) -replace '^#?\s*PasswordAuthentication\s+.*', 'PasswordAuthentication yes' | Set-Content C:\ProgramData\ssh\sshd_config
Start-Service sshd
Start-Sleep 3
Set-NetFirewallProfile -All -Enabled False
New-NetFirewallRule -Name sshd -DisplayName 'OpenSSH Server' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort __PORT__ -ErrorAction SilentlyContinue | Out-Null
Write-Output '--- service ---'
Get-Service sshd | Select-Object Status,StartType | Format-List
Write-Output '--- listening ---'
Get-NetTCPConnection -State Listen -LocalPort __PORT__ -ErrorAction SilentlyContinue | Select-Object LocalAddress,LocalPort | Format-Table
Write-Output '--- config ---'
(Get-Content C:\ProgramData\ssh\sshd_config) -match '^(Port|PasswordAuthentication)'
